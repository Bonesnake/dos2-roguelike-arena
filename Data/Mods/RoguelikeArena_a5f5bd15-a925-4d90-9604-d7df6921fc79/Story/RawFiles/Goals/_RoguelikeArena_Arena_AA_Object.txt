//REGION Events
IF
ObjectEnteredCombat(_Object, _CombatID)
AND
DB_RoguelikeArena_Arenas(_Arena)
AND
QRY_RoguelikeArena_CombatIsInArena(_CombatID, _Arena)
THEN
PROC_RoguelikeArena_ObjectEnteredArenaCombat(_Object, _Arena, _CombatID);

// Triggered when an object (character or item) enters a combat in an arena.
PROC
PROC_RoguelikeArena_ObjectEnteredArenaCombat((GUIDSTRING)_Object, (INTEGER)_Arena, (INTEGER)_CombatID)
THEN
DB_NOOP(1);

IF
ObjectLeftCombat(_Object, _CombatID)
AND
DB_RoguelikeArena_Arenas(_Arena)
AND
QRY_RoguelikeArena_CombatIsInArena(_CombatID, _Arena)
THEN
PROC_RoguelikeArena_ObjectLeftArenaCombat(_Object, _Arena, _CombatID);

// Triggered when an object (character or item) leaves a combat in an arena.
PROC
PROC_RoguelikeArena_ObjectLeftArenaCombat((GUIDSTRING)_Object, (INTEGER)_Arena, (INTEGER)_CombatID)
THEN
DB_NOOP(1);

IF
ObjectTurnStarted(_Object)
AND
DB_RoguelikeArena_ArenaObjects(_Arena, _Object)
THEN
PROC_RoguelikeArena_ObjectArenaTurnStarted(_Object, _Arena);

// Triggered when an object is in combat in an arena, and its current turn has just started.
PROC
PROC_RoguelikeArena_ObjectArenaTurnStarted((GUIDSTRING)_Object, (INTEGER)_Arena)
THEN
DB_NOOP(1);

IF
ObjectTurnEnded(_Object)
AND
DB_RoguelikeArena_ArenaObjects(_Arena, _Object)
THEN
PROC_RoguelikeArena_ObjectArenaTurnEnded(_Object, _Arena);

// Triggered when an object is in combat in an arena, and its current turn has ended.
PROC
PROC_RoguelikeArena_ObjectArenaTurnEnded((GUIDSTRING)_Object, (INTEGER)_Arena)
THEN
DB_NOOP(1);
//END_REGION

THEN
DB_RoguelikeArena_ArenaObjects(_Arena, (GUIDSTRING)_Item);

PROC
PROC_RoguelikeArena_ItemLeftArena(_Item, _Arena)
AND
DB_RoguelikeArena_ArenaObjects(_Arena, _Item)
THEN
NOT DB_RoguelikeArena_ArenaObjects(_Arena, _Item);
//END_REGION

KBSECTION
// Enables arena mode for characters entering an arena.
PROC
PROC_RoguelikeArena_CharacterEnteredArena(_Character, _Arena)
THEN
SetInArena(_Character, 1);

// Disables arena mode for characters exiting an arena.
PROC
PROC_RoguelikeArena_CharacterLeftArena(_Character, _Arena)
AND
CharacterIsDead(_Character, 0)
AND
CharacterIsPolymorphInteractionDisabled(_Character, 0)
THEN
SetInArena(_Character, 0);

// Triggers when an enemy dies.
PROC
PROC_RoguelikeArena_CharacterDiedInArena(_Character, _Arena)
AND
GetFaction(_Character, "Evil NPC")
AND
DB_CombatCharacters(_Character, _CombatID)
AND
NOT QRY_RoguelikeArena_AnyEnemiesInCombat(_CombatID)
// AND
// IsCombatActive(_CombatID, 1)
THEN
PROC_RoguelikeArena_ArenaCombatEnemiesDied(_Arena, _CombatID);

// Triggers when an enemy goes off stage.
PROC
PROC_RoguelikeArena_CharacterWentOnStageInArena(_Character, _Bool, _Arena)
AND
GetFaction(_Character, "Evil NPC")
AND
_Bool == 0
AND
DB_CombatCharacters(_Character, _CombatID)
AND
NOT QRY_RoguelikeArena_AnyEnemiesInCombat(_CombatID)
// AND
// IsCombatActive(_CombatID, 1)
THEN
PROC_RoguelikeArena_ArenaCombatEnemiesDied(_Arena, _CombatID);

// Triggers when an enemy leaves an arena.
PROC
PROC_RoguelikeArena_CharacterLeftArena(_Character, _Arena)
AND
GetFaction(_Character, "Evil NPC")
AND
DB_CombatCharacters(_Character, _CombatID)
AND
NOT QRY_RoguelikeArena_AnyEnemiesInCombat(_CombatID)
// AND
// IsCombatActive(_CombatID, 1)
THEN
PROC_RoguelikeArena_ArenaCombatEnemiesDied(_Arena, _CombatID);

// Triggers when an enemy exits combat.
PROC
PROC_RoguelikeArena_ObjectLeftArenaCombat(_Object, _Arena, _CombatID)
AND
ObjectIsCharacter(_Object, 1)
AND
GetFaction(_Object, "Evil NPC")
AND
NOT QRY_RoguelikeArena_AnyEnemiesInCombat(_CombatID)
// AND
// IsCombatActive(_CombatID, 1)
THEN
PROC_RoguelikeArena_ArenaCombatEnemiesDied(_Arena, _CombatID);

PROC
PROC_RoguelikeArena_ArenaCombatEnemiesDied((INTEGER)_Arena, (INTEGER)_CombatID)
THEN
DB_NOOP(1);

PROC
PROC_RoguelikeArena_EndArenaCombat((INTEGER)_Arena, (INTEGER)_CombatID)
THEN
EndCombat(_CombatID);

PROC
PROC_RoguelikeArena_EndArenaCombat(_Arena, _)
AND
DB_RoguelikeArena_ArenaCharacters(_Arena, _Character)
AND
CharacterIsPlayer(_Character, 1)
THEN
SetInArena(_Character, 0);
PROC_RoguelikeArena_CharacterFullRestore(_Character);

PROC
PROC_RoguelikeArena_EndArenaCombat(_Arena, _CombatID)
THEN
PROC_RoguelikeArena_ArenaCombatEnded(_Arena, _CombatID);

PROC
PROC_RoguelikeArena_ArenaCombatEnded((INTEGER)_Arena, (INTEGER)_CombatID)
THEN
DB_NOOP(1);

//REGION Events
IF
CombatRoundStarted(_CombatID, _Round)
AND
DB_RoguelikeArena_Arenas(_Arena)
AND
QRY_RoguelikeArena_CombatIsInArena(_CombatID, _Arena)
THEN
PROC_RoguelikeArena_ArenaCombatRoundStarted(_Arena, _CombatID, _Round);

// Triggered when a new round has started in an arena.
PROC
PROC_RoguelikeArena_ArenaCombatRoundStarted((INTEGER)_Arena, (INTEGER)_CombatID, (INTEGER)_Round)
THEN
DB_NOOP(1);
//END_REGION

//REGION Queries
// Returns whether any players are currently in combat.
QRY
QRY_RoguelikeArena_AnyPlayersInCombat((INTEGER)_CombatID)
AND
DB_CombatCharacters(_Character, _CombatID)
AND
CharacterIsPlayer(_Character, 1)
AND
CharacterIsDead(_Character, 0)
THEN
DB_NOOP(1);

// Returns whether any enemies are currently in combat.
QRY
QRY_RoguelikeArena_AnyEnemiesInCombat((INTEGER)_CombatID)
AND
DB_CombatCharacters(_Character, _CombatID)
AND
GetFaction(_Character, "Evil NPC")
AND
CharacterIsDead(_Character, 0)
THEN
DB_NOOP(1);

// Returns whether combat with _CombatID is inside _Arena.
QRY
QRY_RoguelikeArena_CombatIsInArena((INTEGER)_CombatID, (INTEGER)_Arena)
AND
DB_CombatCharacters(_Character, _CombatID)
AND
DB_RoguelikeArena_ArenaCharacters(_Arena, _Character)
THEN
DB_NOOP(1);
//END_REGION
